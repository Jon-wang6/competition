# 裁判规则审计专用C++程序；不链接或安装到选手决策进程。
cmake_minimum_required(VERSION 3.16)
project(referee_rules_audit)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(simulator_probe simulator_probe.cpp)
target_compile_features(simulator_probe PRIVATE cxx_std_17)
ament_target_dependencies(simulator_probe rclcpp std_msgs)
target_link_libraries(simulator_probe nlohmann_json::nlohmann_json)
add_executable(public_probe public_probe.cpp)
target_compile_features(public_probe PRIVATE cxx_std_17)
ament_target_dependencies(public_probe rclcpp std_msgs)
target_link_libraries(public_probe nlohmann_json::nlohmann_json)
